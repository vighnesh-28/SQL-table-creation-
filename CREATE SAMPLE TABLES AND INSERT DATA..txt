QUESTION 5 :

     display departments that have at least one employee.
--->
  SELECT DepartmentName
  FROM Departments d
  WHERE EXISTS (
  SELECT 1 FROM Employees e
  WHERE e.DepartmentID=d.DepartmentID);

QUESTION 6 :
